## PlatformHardgateStatus schema

```python
from pydantic import BaseModel
from typing import Literal

class PlatformHardgateStatus(BaseModel):
    phase_id: str
    baseline_version: str = "v3.5.20"
    real_trade_allowed: bool = False
    broker_order_allowed: bool = False
    runtime_enabled: bool = False
    auto_buy_allowed: bool = False
    auto_sell_allowed: bool = False
    production_strategy_modified: bool = False
    classifier_production_modified: bool = False
    max_promotion_status_allowed: Literal["RESEARCH_READY", "PAPER_ONLY", "NO_PRODUCTION"] = "NO_PRODUCTION"
    baseline_rewritten: bool = False
```
