## BMatrixCurrentSnapshotItem

```python
from pydantic import BaseModel
from typing import Literal

class BMatrixCurrentSnapshotItem(BaseModel):
    snapshot_id: str
    ticker: str
    trade_date: str
    data_as_of_date: str
    financial_report_date: str | None = None
    source_dataset: str
    pit_status: Literal["CURRENT_SNAPSHOT_ONLY", "PIT_SAFE", "PIT_BLOCKED", "PIT_UNKNOWN"]
    usable_for_backtest: bool = False
    usable_for_current_snapshot: bool = True
    historical_return_claim_allowed: bool = False
    production_allowed: bool = False
    real_trade_allowed: bool = False
    evidence_status: Literal["READY", "DATA_INSUFFICIENT", "STALE", "CONFLICTED"]
```
